# src/core/hand_analyzer.py 末尾追加
def can_peng(hand: List[Tile], tile: Tile) -> bool:
    """True if the hand holds at least two copies of `tile` (can peng it)."""
    return sum(1 for t in hand if t == tile) >= 2


def can_chi(hand: List[Tile], tile: Tile) -> List[List[Tile]]:
    """All chi combos for claiming a discarded number tile.

    Each entry is the TWO tiles from hand that complete the run with `tile`
    (covers edge and middle-gap shapes). Honors return [].
    """
    if tile.suit not in ("万", "饼", "条"):
        return []
    v = tile.value
    combos = []
    for x, y in ((v - 2, v - 1), (v - 1, v + 1), (v + 1, v + 2)):
        if 1 <= x and y <= 9:
            ta, tb = create_tile(tile.suit, x), create_tile(tile.suit, y)
            if ta in hand and tb in hand:
                combos.append([ta, tb])
    return combos


def can_ron(hand: List[Tile], tile: Tile, fuuro: int = 0) -> bool:
    """True if adding `tile` to the hand completes a winning hand."""
    return calculate_shanten(hand + [tile], fuuro) == -1

PHASE_NEED_DISCARD = "need_discard"

_PLAYER_ZH = {"me": "我", "left": "上家", "right": "下家", "opposite": "对家"}

_PLAYERS = {
    "上家": "left", "left": "left", "l": "left",

_MY_ACTIONS = [
    ("杠花", "kong_flower"), ("kg", "kong_flower"),
    ("暗杠", "ankan"), ("an", "ankan"),
    ("加杠", "kakan"), ("ka", "kakan"),

@dataclass
class Event:
    kind: str
    player: str = "me"
    tiles: List[Tile] = field(default_factory=list)
    arg: Optional[str] = None            # 胡 放炮/自摸/杠开 用


def parse_event(line: str) -> Optional[Event]:
    """Parse one session input line into an Event; None for blank input."""
    s = line.strip()
    if not s:
        return None
    if s in ("q", "quit", "退", "退出"):
        return Event("quit")
    if s in ("看", "分析"):
        return Event("analyze")
    if s in ("撤", "z", "undo"):
        return Event("undo")
    if s in ("局", "局面", "状态"):
        return Event("status")
    if s in ("过", "pass"):
        return Event("pass")

    for kw, kind in _MY_ACTIONS:
        if s.startswith(kw):
            names = [t for t in s[len(kw):].strip().split() if t]
            tiles = [parse_tile(t) for t in names]
            if not tiles:
                raise ValueError(f"{kw!r} 后面缺少牌")
            return Event(kind, "me", tiles)

    for pref in sorted(_PLAYERS, key=len, reverse=True):
        if s.startswith(pref):
            rest = s[len(pref):].strip()
            if not rest:
                raise ValueError(f"{pref} 后面缺少动作")
            player = _PLAYERS[pref]
            if rest[0] == "碰":
                return Event("peng", player, [parse_tile(t) for t in rest[1:].split()])
            if rest[0] == "吃":
                return Event("chi", player, [parse_tile(t) for t in rest[1:].split()])
            if rest[0] == "杠":
                return Event("gang", player, [parse_tile(t) for t in rest[1:].split()])
            if rest[0] == "打":
                rest = rest[1:].strip()
            return Event("discard", player, [parse_tile(t) for t in rest.split()])

    raise ValueError(f"无法识别 {s!r}（输入 帮助 查看命令）")


@dataclass
class GameSession:
    """事件驱动的对局状态机：阶段 + 待响应弃牌 + 摸牌位 + 胡牌记录 + 快照撤销。"""

    state: GameState = field(default_factory=GameState)
    phase: str = PHASE_NEED_DRAW
    discard_seq: int = 0
    pending: Optional[dict] = None        # {"id": int, "player": str, "tile": Tile}
    drawn_tile: Optional[Tile] = None
    win: Optional[dict] = None            # {"method","tile","source"}
    konged_this_turn: bool = False
    history: list = field(default_factory=list)

    def __post_init__(self):
        if not self.history:
            self.history = [self._snap()]

    # ---------- 基础 ----------
    def base(self) -> int:
        return 13 - 3 * len(self.state.my_melds)

    def _snap(self):
        return copy.deepcopy((self.state, self.phase, self.discard_seq,
                              self.pending, self.drawn_tile, self.win,
                              self.konged_this_turn))

    def _restore(self, snap):
        (self.state, self.phase, self.discard_seq, self.pending,
         self.drawn_tile, self.win, self.konged_this_turn) = snap

    # ---------- 对外入口 ----------
    def apply(self, line: str) -> dict:
        try:
            ev = parse_event(line)
        except ValueError as e:
            return {"ok": False, "error": str(e)}
        if ev is None:
            return {"ok": False, "error": "空输入"}
        if ev.kind == "undo":
            return self.undo()
        if ev.kind == "quit":
            return {"ok": False, "error": "命令行模式 Ctrl+C 退出；网页模式关页面即可"}
        if ev.kind in ("analyze", "status"):
            return {"ok": True, "noop": ev.kind}
        try:
            self._dispatch(ev)
        except ValueError as e:
            return {"ok": False, "error": str(e)}
        self.history.append(self._snap())
        return {"ok": True, "kind": ev.kind}

    def undo(self) -> dict:
        if len(self.history) > 1:
            self.history.pop()
            self._restore(self.history[-1])
            return {"ok": True, "undid": True}
        return {"ok": False, "error": "没有可撤销的事件"}

    def _dispatch(self, ev: Event):
        if ev.kind == "draw":
            self._ev_draw(ev)
        elif ev.kind == "discard":
            self._ev_discard(ev)
        elif ev.kind == "pass":
            self._ev_pass(ev)
        else:
            raise ValueError(f"暂不支持的操作：{ev.kind}")

    # ---------- 待响应弃牌 ----------
    def _remaining(self, tile: Tile) -> int:
        return count_remaining_tiles(self.state).get(tile, 0)

    def _set_pending(self, player: str, tile: Tile):
        self.discard_seq += 1
        self.pending = {"id": self.discard_seq, "player": player, "tile": tile}

    def _require_pending(self, tile: Tile):
        if self.pending is None:
            raise ValueError("没有可响应的弃牌（先记录那张牌被打出）")
        if self.pending["tile"] != tile:
            raise ValueError(
                f"待响应的弃牌是 {self.pending['tile']}"
                f"（来自{_PLAYER_ZH[self.pending['player']]}），不是 {tile}")

    def _take_pending_tile(self):
        river = (self.state.my_discards if self.pending["player"] == "me"
                 else self.state.discards[self.pending["player"]])
        river.remove(self.pending["tile"])
        self.pending = None

    # ---------- 摸/打/过 ----------
    def _ev_draw(self, ev: Event):
        tile = ev.tiles[0]
        if self.phase != PHASE_NEED_DRAW or len(self.state.my_hand) != self.base():
            raise ValueError("现在不是摸牌时机（手牌未打出或已摸过）")
        if self._remaining(tile) <= 0:
            raise ValueError(f"{tile} 已全部可见，请检查输入")
        if tile.is_flower:
            self.state.flowers_in_hand.append(tile)
        else:
            self.state.my_hand.append(tile)
        self.drawn_tile = tile

    def _ev_discard(self, ev: Event):
        tile = ev.tiles[0]
        if ev.player == "me":
            if self.phase != PHASE_NEED_DISCARD or len(self.state.my_hand) != self.base() + 1:
                raise ValueError("先摸牌（或碰/吃后）才能打牌")
            if tile.is_flower:
                raise ValueError("花牌不能打出")
            if tile not in self.state.my_hand:
                raise ValueError(f"手里没有 {tile}")
            self.state.my_hand.remove(tile)
            self.state.my_discards.append(tile)
            self.drawn_tile = None
            self.konged_this_turn = False
            self._set_pending("me", tile)
            self.phase = PHASE_NEED_DRAW
        else:
            if self._remaining(tile) <= 0:
                raise ValueError(f"{tile} 已全部可见，请检查输入")
            self.state.discards[ev.player].append(tile)
            self._set_pending(ev.player, tile)

    def _ev_pass(self, ev: Event):
        self.pending = None


def new_session(hand_line: str = "", flowers_line: str = "") -> GameSession:
    hand = [parse_tile(t) for t in hand_line.split() if t]
    flowers = [parse_tile(t) for t in flowers_line.split() if t]
    return GameSession(state=GameState(my_hand=hand, flowers_in_hand=flowers))

    if state.my_discards:

        print_fn(f"{label}弃牌：{river} | 明牌：{melds}")


def run_session(session: GameSession, input_fn=input, print_fn=print) -> GameSession:
    """命令行会话循环：事件进，摸牌后出分析。"""
    print_fn(_HELP)
    while True:
        try:
            line = input_fn("\n事件 > ")
        except EOFError:
            break
        result = session.apply(line)
        if not result.get("ok"):
            if result.get("error"):
                print_fn(f"✗ {result['error']}")
            continue
        if result.get("undid"):
            print_fn("已撤销上一步")
            print_status(session, print_fn)
            continue
        if result.get("noop") == "analyze":
            print_analysis(analyze(session.state), print_fn)
            continue

            continue
        if result.get("kind") == "draw" and session.phase == PHASE_NEED_DISCARD:
            print_analysis(analyze(session.state), print_fn)
    return session

    def _ev_kong_flower(self, ev: Event):
        tile = ev.tiles[0]
        if tile not in self.state.flowers_in_hand:
            raise ValueError(f"未杠花牌里没有 {tile}")
        if tile.suit == "季节":
            self.state.flowers_in_hand.remove(tile)
            self.state.konged_flowers.append(tile)
        elif sum(1 for f in self.state.flowers_in_hand if f == tile) >= 4:
            for _ in range(4):
                self.state.flowers_in_hand.remove(tile)
            self.state.konged_flowers.extend([tile] * 4)

def validate_new_game(hand: List[Tile], flowers: List[Tile]) -> GameState:
    """开局校验：13 张暗牌、逐种不超副本上限、花牌不混入暗牌。"""
    if len(hand) != 13:
        raise ValueError(f"起手暗牌应为 13 张，当前 {len(hand)} 张")
    for t in hand:
        if t.is_flower:
            raise ValueError(f"{t} 是花牌，请放在花牌区")
    counts: dict = {}
    for t in hand:
        counts[t] = counts.get(t, 0) + 1
    for t, n in counts.items():
        if n > 4:
            raise ValueError(f"{t} 超过 4 张（{n}）")
    fc: dict = {}
    for t in flowers:
        if not t.is_flower:
            raise ValueError(f"{t} 不是花牌")
        fc[t] = fc.get(t, 0) + 1
    for t, n in fc.items():
        limit = 1 if t.suit == "季节" else 4
        if n > limit:
            raise ValueError(f"{t} 超过 {limit} 张（{n}）")
    return GameState(my_hand=list(hand), flowers_in_hand=list(flowers))
